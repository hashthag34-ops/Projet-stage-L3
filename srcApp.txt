import { BrowserRouter as Router, Routes, Route } from 'react-router-dom';

// Pages simples pour démarrer

const Login = () => <div className="p-8 text-2xl font-bold">Page de Connexion 🔐</div>;
const Dashboard = () => <div className="p-8 text-2xl font-bold">Espace Apprenant 🎓</div>;

function App() {
  return (
    <Router>
      <div className="min-h-screen bg-gray-50 text-gray-800">
        <Routes>
          {/* Espace Public */}
          <Route path="/" element={<Home />} />
          <Route path="/login" element={<Login />} />
          
          {/* Espace Privé */}
          <Route path="/dashboard" element={<Dashboard />} />
        </Routes>
      </div>
    </Router>
  );
}

export default App;
